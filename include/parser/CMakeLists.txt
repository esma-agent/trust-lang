# include/parser/CMakeLists.txt
# Parser library — lexer (Flex) + parser (Bison) + macro processor (MMProc)

set(GEN_DIR "${CMAKE_BINARY_DIR}/include/parser")

# ── Flex: generate lexer from lexer.l ──
set(LEXER_L "${CMAKE_CURRENT_SOURCE_DIR}/lexer.l")

add_custom_command(
    OUTPUT "${GEN_DIR}/lex.yy.cpp" "${GEN_DIR}/flex.gen.h"
    COMMAND ${FLEX_EXECUTABLE} -w
        --outfile=${GEN_DIR}/lex.yy.cpp
        --header-file=${GEN_DIR}/flex.gen.h
        ${LEXER_L}
    DEPENDS "${LEXER_L}"
    COMMENT "[Flex] Generating lexer from lexer.l..."
    VERBATIM
)

# ── Token generation header (generated by ast component) ──
# The generated token files are in ${CMAKE_BINARY_DIR}/include/ast/
# They are a dependency of ast_lib

# ── Bison: generate parser from parser.gen.y (generated by ast component) ──
add_custom_command(
    OUTPUT "${GEN_DIR}/parser.tab.cpp" "${GEN_DIR}/parser.tab.hh"
    COMMAND ${BISON_EXECUTABLE}
        --report=all
        --defines=${GEN_DIR}/parser.tab.hh
        --output=${GEN_DIR}/parser.tab.cpp
        "${CMAKE_BINARY_DIR}/include/ast/parser.gen.y"
    DEPENDS
        "${CMAKE_BINARY_DIR}/include/ast/parser.gen.y"
    COMMENT "[Bison] Generating parser from parser.gen.y..."
    VERBATIM
)

# ── Generated files target ──
add_custom_target(parser_generated ALL
    DEPENDS
        "${GEN_DIR}/lex.yy.cpp"
        "${GEN_DIR}/flex.gen.h"
        "${GEN_DIR}/parser.tab.cpp"
        "${GEN_DIR}/parser.tab.hh"
)
add_dependencies(parser_generated ast_generated)

# ── Parser library ──
add_library(parser_lib STATIC
    ${PROJECT_SOURCE_DIR}/parser/mmproc.cpp
    ${PROJECT_SOURCE_DIR}/parser/parser.cpp
    ${GEN_DIR}/lex.yy.cpp
    ${GEN_DIR}/parser.tab.cpp
)
target_link_libraries(parser_lib PUBLIC ast_lib diag_lib)
target_include_directories(parser_lib PUBLIC
    ${CMAKE_BINARY_DIR}/include
    ${PROJECT_INCLUDE_DIR}
    ${GEN_DIR}
)
set_source_files_properties(
    ${PROJECT_SOURCE_DIR}/parser/mmproc.cpp
    ${GEN_DIR}/lex.yy.cpp
    ${GEN_DIR}/parser.tab.cpp
    PROPERTIES COMPILE_OPTIONS "-Wno-old-style-cast"
)
add_dependencies(parser_lib parser_generated ast_generated generate_dsl_embed)
